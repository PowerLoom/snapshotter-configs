# https://ipfs.io/ipfs/bafkreicdywe3fj2zsjfdqbdfkf6scjwwxfhnag4exahyha3bmhbuwzcpfq
import requests
from concurrent.futures import ThreadPoolExecutor, as_completed
import logging
import time

# Set up logging
logging.basicConfig(level=logging.INFO, 
                   format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)


def fetch_ipfs_data(cid):
    url = f"https://ipfs.io/ipfs/{cid}"
    try:
        response = requests.get(url, timeout=30)
        
        # Check if request was successful
        if response.status_code != 200:
            logger.error(f"HTTP {response.status_code} for CID {cid}: "
                         f"{response.text[:200]}")
            raise Exception(f"HTTP {response.status_code}: {response.text[:100]}")
        
        # Check if response has content
        if not response.content:
            logger.error(f"Empty response for CID {cid}")
            raise Exception("Empty response from IPFS")
        
        # Log response info
        content_type = response.headers.get('Content-Type', 'unknown')
        logger.debug(f"Response for {cid}: {len(response.content)} bytes, "
                     f"Content-Type: {content_type}")
        
        # Try to parse JSON
        try:
            data = response.json()
            return data
        except ValueError as e:
            logger.error(f"JSON parsing error for CID {cid}: {e}")
            logger.error(f"Response content (first 500 chars): "
                         f"{response.text[:500]}")
            raise Exception(f"Invalid JSON response: {e}")
            
    except requests.exceptions.RequestException as e:
        logger.error(f"Request error for CID {cid}: {e}")
        raise Exception(f"Request failed: {e}")


def collect_snapshot_chain(initial_cid, target_epoch):
    """Collect all CIDs in the snapshot chain down to target epoch"""
    cids_to_fetch = []
    current_cid = initial_cid
    
    logger.info(f"Starting snapshot chain collection from {initial_cid}")
    logger.info(f"Target epoch: {target_epoch}")
    
    # First, we need to sequentially follow the chain to collect all CIDs
    iteration = 0
    while True:
        iteration += 1
        logger.info(f"Chain iteration {iteration}: Processing CID {current_cid}")
        
        try:
            # Fetch current snapshot to get the next CID in chain
            data = fetch_ipfs_data(current_cid)
            cids_to_fetch.append(current_cid)
            
            last_epoch = data["epoch"]["begin"]
            logger.info(f"Last epoch: {last_epoch}, Current CID: {current_cid}")
            
            updated = False
            if "previousSnapshots" in data and data["previousSnapshots"]:
                prev_count = len(data['previousSnapshots'])
                logger.info(f"Found {prev_count} previous snapshots")
                
                for epoch_id, cid in data["previousSnapshots"][::-1]:
                    logger.debug(f"Checking epoch {epoch_id} (target: {target_epoch})")
                    if epoch_id >= target_epoch:
                        cids_to_fetch.append(cid)
                        last_epoch = epoch_id
                        logger.info(f"Last epoch: {last_epoch}, Current CID: {cid}")
                        updated = True
                    else:
                        logger.debug(f"Skipping epoch {epoch_id} (below target)")
            else:
                logger.warning(f"No previousSnapshots found in data for CID "
                              f"{current_cid}")

            if not updated:
                logger.info(f"Current epoch: {last_epoch} is less than target "
                           f"epoch: {target_epoch}, stopping chain collection")
                break
             
            current_cid = cids_to_fetch[-1]
            logger.info(f"Next CID in chain: {current_cid}")
            
            # Add small delay to avoid overwhelming IPFS gateway
            time.sleep(0.1)
            
        except Exception as e:
            logger.error(f"Error processing CID {current_cid} in chain "
                        f"collection: {e}")
            break
    
    logger.info(f"Chain collection completed. Found {len(cids_to_fetch)} "
               f"total CIDs")
    return cids_to_fetch


def fetch_all_snapshots_parallel(cids_to_fetch):
    """Fetch all snapshots in parallel"""
    logger.info(f"Starting parallel fetch of {len(cids_to_fetch)} snapshots")
    cids_to_fetch = list(set(cids_to_fetch))
    all_snapshot_data = {}
    failed_cids = []
    
    # Reduced workers to be gentler on IPFS
    with ThreadPoolExecutor(max_workers=5) as executor:
        # Submit all fetch tasks
        future_to_cid = {executor.submit(fetch_ipfs_data, cid): cid 
                        for cid in cids_to_fetch}
        
        # Collect results as they complete
        completed = 0
        for future in as_completed(future_to_cid):
            cid = future_to_cid[future]
            completed += 1
            try:
                data = future.result()
                all_snapshot_data[cid] = data
            except Exception as exc:
                logger.error(f"CID {cid} generated an exception: {exc}")
                failed_cids.append(cid)
    
    if failed_cids:
        logger.warning(f"Failed to fetch {len(failed_cids)} CIDs: {failed_cids}")
    
    success_count = len(all_snapshot_data)
    total_count = len(cids_to_fetch)
    logger.info(f"Parallel fetch completed. Successfully fetched "
               f"{success_count}/{total_count} snapshots")
    return all_snapshot_data
